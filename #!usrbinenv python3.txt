#!/usr/bin/env python3
import argparse, base64, getpass, json, os, sys, time
from pathlib import Path

from fastapi.responses import HTMLResponse, StreamingResponse

BASE = Path(__file__).parent
SESSION_FILE = BASE / "session"
IG_USER = os.environ.get("IG_USERNAME", "").strip()
CACHE_TTL, STORY_TTL = 300, 60
UA = ("Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
      "(KHTML, like Gecko) Chrome/122.0 Safari/537.36")
ALLOWED = (".cdninstagram.com", ".fbcdn.net", "instagram.com")

L = instaloader.Instaloader(quiet=True, download_video_thumbnails=False,
                            save_metadata=False, post_metadata_txt_pattern="",
                            max_connection_attempts=2)
_cache: dict[str, tuple[float, Any]] = {}

def restore_session():
    raw = os.environ.get("IG_SESSION")
    if raw and not SESSION_FILE.exists():
        try:
            SESSION_FILE.write_bytes(base64.b64decode(raw))
        except Exception as e:
            print("session restore failed:", e, file=sys.stderr)

def cached(key, fn, ttl=CACHE_TTL):
    hit = _cache.get(key)
    if hit and time.time() - hit[0] < ttl:
        return hit[1]
    val = fn()
    _cache[key] = (time.time(), val)
    return val

def ctx():
    restore_session()
    if not SESSION_FILE.exists():
        raise HTTPException(503, "no session - run: python app.py --login")
    if not L.context.is_logged_in:
        L.load_session_from_file(IG_USER, str(SESSION_FILE))
    return L

def get_profile(username: str) -> instaloader.Profile:
    try:
        return instaloader.Profile.from_username(ctx().context, username)
    except instaloader.exceptions.ProfileNotExistsException:
        raise HTTPException(404, f"@{username} does not exist")

def blocked(p) -> bool:
    return p.is_private and not p.followed_by_viewer

def ser(node, kind="post") -> dict:
    vid = node.is_video
    out = {"id": node.shortcode if kind == "post" else str(node.mediaid), "kind": kind,
           "is_video": vid, "thumb": node.url, "url": node.video_url if vid else node.url,
           "caption": (node.caption or "") if kind == "post" else "",
           "likes": getattr(node, "likes", None), "comments": getattr(node, "comments", None),
           "taken_at": node.date_utc.isoformat(), "carousel": []}
    if getattr(node, "typename", "") == "GraphSidecar":
        for n in node.get_sidecar_nodes():
            out["carousel"].append({"is_video": n.is_video,
                                    "url": n.video_url if n.is_video else n.url, "thumb": n.url})
    return out

app = FastAPI(title="ig-viewer", docs_url=None, redoc_url=None)

@app.get("/api/health")
def health():
    restore_session()
    return {"ok": SESSION_FILE.exists(), "session_account": IG_USER}

@app.get("/api/profile/{username}")
def profile(username: str):
    def f():
        p = get_profile(username)
        return {"username": p.username, "full_name": p.full_name, "biography": p.biography,
                "followers": p.followers, "followees": p.followees, "post_count": p.mediacount,
                "is_private": p.is_private, "is_verified": p.is_verified,
                "profile_pic": p.profile_pic_url, "external_url": p.external_url,
                "following": p.followed_by_viewer, "requested": p.has_requested_viewer,
                "locked": blocked(p)}
    return cached(f"p:{username}", f)

@app.post("/api/follow/{username}")
def follow(username: str):
    p = get_profile(username)
    if not p.followed_by_viewer:
        try:
            p.follow()
        except Exception as e:
            raise HTTPException(400, str(e))
    _cache.clear()
    return {"following": True, "requested": p.is_private}

@app.get("/api/posts/{username}")
def posts(username: str, limit: int = Query(24, le=100)):
    def f():
        p = get_profile(username)
        if blocked(p):
            return {"items": [], "locked": True}
        items = []
        for n in p.get_posts():
            items.append(ser(n))
            if len(items) >= limit:
                break
        return {"items": items, "locked": False}
    return cached(f"posts:{username}:{limit}", f)

@app.get("/api/stories/{username}")
def stories(username: str):
    def f():
        p = get_profile(username)
        if blocked(p):
            return {"items": [], "locked": True}
        items = [ser(i, "story") for s in ctx().get_stories(userids=[p.userid]) for i in s.get_items()]
        items.sort(key=lambda x: x["taken_at"])
        return {"items": items, "locked": False}
    return cached(f"st:{username}", f, STORY_TTL)

@app.get("/api/highlights/{username}")
def highlights(username: str):
    def f():
        p = get_profile(username)
        if blocked(p):
            return {"items": [], "locked": True}
        return {"items": [{"id": h.unique_id, "title": h.title, "cover": h.cover_url,
                           "count": h.itemcount} for h in ctx().get_highlights(p)], "locked": False}
    return cached(f"hl:{username}", f)

@app.get("/api/highlight/{unique_id}")
def highlight_items(unique_id: str):
    def f():
        hl = instaloader.Highlight.from_id(ctx().context, int(unique_id))
        return {"title": hl.title, "items": [ser(i, "story") for i in hl.get_items()]}
    return cached(f"hli:{unique_id}", f)

@app.get("/api/reposts/{username}")
def reposts(username: str, limit: int = Query(24, le=100)):
    def f():
        p = get_profile(username)
        if blocked(p):
            return {"items": [], "locked": True}
        raw = ctx().context.get_raw(
            f"https://www.instagram.com/api/v1/feed/user/{p.userid}/reposts/?count={limit}")
        out = []
        for entry in json.loads(raw).get("items", []):
            m = entry.get("media", entry)
            car = []
            for c in m.get("carousel_media") or []:
                cand = (c.get("image_versions2", {}).get("candidates") or [{}])[0].get("url")
                vid = (c.get("video_versions") or [{}])[0].get("url")
                car.append({"is_video": c.get("media_type") == 2, "url": vid or cand, "thumb": cand})
            cand = (m.get("image_versions2", {}).get("candidates") or [{}])[0].get("url")
            vid = (m.get("video_versions") or [{}])[0].get("url")
            out.append({"id": m.get("code") or m.get("pk"), "kind": "repost",
                        "is_video": m.get("media_type") == 2, "thumb": cand, "url": vid or cand,
                        "caption": ((m.get("caption") or {}).get("text")) or "",
                        "likes": m.get("like_count"), "comments": m.get("comment_count"),
                        "taken_at": time.strftime("%Y-%m-%dT%H:%M:%S", time.gmtime(m.get("taken_at", 0))),
                        "carousel": car})
        return {"items": out, "locked": False}
    return cached(f"rp:{username}:{limit}", f)

@app.get("/api/media")
async def media(url: str):
    host = urlparse(url).netloc
    if not any(host == a.lstrip(".") or host.endswith(a) for a in ALLOWED):
        raise HTTPException(400, "host not allowed")
    client = httpx.AsyncClient(follow_redirects=True, timeout=30)
    up = await client.send(client.build_request(
        "GET", url, headers={"Referer": "https://www.instagram.com/", "User-Agent": UA}), stream=True)
    if up.status_code != 200:
        await up.aclose(); await client.aclose()
        raise HTTPException(up.status_code, "upstream failed")
    hdrs = {"Cache-Control": "public, max-age=86400"}
    for h in ("content-type", "content-length", "accept-ranges"):
        if h in up.headers:
            hdrs[h] = up.headers[h]
    async def done():
        await up.aclose(); await client.aclose()
    return StreamingResponse(up.aiter_raw(), headers=hdrs, background=done)

def do_login():
    user = IG_USER or input("burner IG username: ").strip()
    L2 = instaloader.Instaloader(quiet=True)
    try:
        L2.login(user, getpass.getpass("password: "))
    except instaloader.exceptions.TwoFactorAuthRequiredException:
        L2.two_factor_login(input("2FA code: ").strip())
    L2.save_session_to_file(str(SESSION_FILE))
    print("session saved")

if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--login", action="store_true")
    ap.add_argument("--port", type=int, default=int(os.environ.get("PORT", 8000)))
    a = ap.parse_args()
    if a.login:
        do_login()
        sys.exit(0)
    restore_session()
    uvicorn.run(app, host="0.0.0.0", port=a.port, log_level="warning")
